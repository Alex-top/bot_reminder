 Детальная структура папок и файлов
 
reminder-bot/
│
├── .env                          # переменные окружения (токен, ID группы)
├── .gitignore                    # игнорируем venv, .env, базу данных
├── requirements.txt              # зависимости проекта
├── README.md                     # описание проекта (для портфолио)
│
├── bot.py                        # точка входа, запуск бота
├── config.py                     # загрузка конфигурации из .env
│
├── models/
│   ├── __init__.py
│   └── reminder.py               # класс Reminder (модель данных)
│
├── handlers/
│   ├── __init__.py
│   ├── message_handler.py        # обработка текстовых сообщений
│   ├── command_handler.py        # обработка команд (/start, /help, /list, /cancel)
│   └── callback_handler.py       # обработка callback-кнопок (если будут)
│
├── services/
│   ├── __init__.py
│   ├── reminder_service.py       # бизнес-логика работы с напоминаниями
│   ├── message_service.py        # отправка сообщений пользователям
│   └── scheduler_service.py      # планировщик (фоновый поток)
│
├── parsers/
│   ├── __init__.py
│   ├── time_parser.py            # основной парсер (координирует regex и hors)
│   ├── regex_parser.py           # парсинг через регулярные выражения
│   └── hors_parser.py            # парсинг через библиотеку hors
│
├── storage/
│   ├── __init__.py
│   ├── database.py               # инициализация БД, подключение
│   └── reminder_repository.py    # CRUD операции с напоминаниями
│
├── utils/
│   ├── __init__.py
│   ├── logger.py                 # настройка логирования
│   ├── validators.py             # валидация входящих данных
│   └── formatters.py             # форматирование дат, сообщений
│
└── tests/                        # (опционально, для портфолио — плюс)
    ├── __init__.py
    ├── test_parsers.py
    └── test_repository.py
